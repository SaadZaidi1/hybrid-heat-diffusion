cmake_minimum_required(VERSION 3.16)
project(heat_diffusion LANGUAGES CXX)
find_package(MPI REQUIRED COMPONENTS CXX)
find_package(OpenMP REQUIRED)
add_executable(heat src/main.cpp)
target_compile_features(heat PRIVATE cxx_std_17)
target_link_libraries(heat PRIVATE MPI::MPI_CXX OpenMP::OpenMP_CXX)
target_compile_options(heat PRIVATE -O3 -Wall -Wextra)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(heat PRIVATE -march=native)
endif()
enable_testing()
add_test(NAME sequential COMMAND heat --size 128 --steps 20 --mode seq_naive --verify)
